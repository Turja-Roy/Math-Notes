\section*{3D Space \& Vectors}

\subsection*{Line Equations}
\textbf{Vector:} $\vec{r} = \vec{r_0} + t\vec{v}$ \\
\textbf{Parametric:} $x = x_0 + ta, y = y_0 + tb, z = z_0 + tc$ \\
\textbf{Symmetric:} $\dfrac{x-x_0}{a} = \dfrac{y-y_0}{b} = \dfrac{z-z_0}{c}$

\subsection*{Plane Equations}
\textbf{Vector:} $\vec{n} \cdot (\vec{r} - \vec{r_0}) = 0$ \\
\textbf{Scalar:} $a(x-x_0) + b(y-y_0) + c(z-z_0) = 0$ \\
\textbf{General:} $ax + by + cz = d$ where $d = ax_0 + by_0 + cz_0$

\subsection*{Quadratic Surfaces}
\textbf{Ellipsoid:} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1$ \\
\textbf{Cone:} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = \dfrac{z^2}{c^2}$ \\
\textbf{Cylinder:} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ \\
\textbf{Hyperboloid (1 sheet):} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} - \dfrac{z^2}{c^2} = 1$ \\
\textbf{Hyperboloid (2 sheets):} $-\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1$ \\
\textbf{Elliptic Paraboloid:} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = \dfrac{z}{c}$ \\
\textbf{Hyperbolic Paraboloid:} $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = \dfrac{z}{c}$

\subsection*{Vector Functions}
$\vec{r}(t) = \langle f(t), g(t), h(t) \rangle$ \\
$\vec{r}'(t) = \langle f'(t), g'(t), h'(t) \rangle$ (tangent vector) \\
$\vec{T}(t) = \dfrac{\vec{r}'(t)}{\|\vec{r}'(t)\|}$ (unit tangent) \\
$\vec{N}(t) = \dfrac{\vec{T}'(t)}{\|\vec{T}'(t)\|}$ (unit normal) \\
$\vec{B}(t) = \vec{T}(t) \times \vec{N}(t)$ (binormal) \\
\textbf{Arc Length:} $L = \int_a^b \|\vec{r}'(t)\| \dd{t}$ \\
\textbf{Curvature:} $\kappa = \dfrac{\|\vec{T}'(t)\|}{\|\vec{r}'(t)\|} = \dfrac{\|\vec{r}'(t) \times \vec{r}''(t)\|}{\|\vec{r}'(t)\|^3}$

\section*{Partial Derivatives}

\subsection*{Notation}
$f_x = \pdv{f}{x}, \quad f_y = \pdv{f}{y}, \quad f_z = \pdv{f}{z}$ \\
$f_{xx} = \pdv[2]{f}{x}, \quad f_{xy} = \pdv{f}{y,x}, \quad f_{yy} = \pdv[2]{f}{y}$

\textbf{Clairaut's Theorem:} $f_{xy} = f_{yx}$ (if continuous)

\subsection*{Differentials}
$\dd{f} = f_x \dd{x} + f_y \dd{y} + f_z \dd{z}$

\subsection*{Chain Rule}
If $z = f(x,y)$, $x = g(t)$, $y = h(t)$: \\
$\dfrac{dz}{dt} = \pdv{f}{x}\dfrac{dx}{dt} + \pdv{f}{y}\dfrac{dy}{dt}$

If $z = f(x,y)$, $x = g(s,t)$, $y = h(s,t)$: \\
$\pdv{z}{s} = f_x\pdv{x}{s} + f_y\pdv{y}{s}$ \\
$\pdv{z}{t} = f_x\pdv{x}{t} + f_y\pdv{y}{t}$

\textbf{General:} $\pdv{z}{t_i} = \sum_{j=1}^n \pdv{z}{x_j}\pdv{x_j}{t_i}$

\subsection*{Implicit Differentiation}
If $F(x,y) = 0$: $\dfrac{dy}{dx} = -\dfrac{F_x}{F_y}$

If $F(x,y,z) = 0$: $\pdv{z}{x} = -\dfrac{F_x}{F_z}, \quad \pdv{z}{y} = -\dfrac{F_y}{F_z}$

\subsection*{Directional Derivatives \& Gradient}
\textbf{Gradient:} $\nabla f = \langle f_x, f_y, f_z \rangle$ \\
\textbf{Directional Derivative:} $D_{\vec{u}}f = \nabla f \cdot \vec{u}$ \\
Max rate of change: $\|\nabla f\|$ in direction of $\nabla f$ \\
$\nabla f$ is orthogonal to level curves/surfaces

\subsection*{Tangent Planes \& Linear Approximation}
\textbf{Tangent Plane to $z=f(x,y)$ at $(x_0,y_0,z_0)$:} \\
$z = f(x_0,y_0) + f_x(x_0,y_0)(x-x_0) + f_y(x_0,y_0)(y-y_0)$

\textbf{Tangent Plane to $F(x,y,z)=k$ at $(x_0,y_0,z_0)$:} \\
$F_x(x-x_0) + F_y(y-y_0) + F_z(z-z_0) = 0$

\textbf{Normal Line:} $\vec{r}(t) = \langle x_0,y_0,z_0 \rangle + t\nabla F(x_0,y_0,z_0)$

\section*{Optimization}

\subsection*{Critical Points}
$\nabla f = \vec{0}$ or $\nabla f$ does not exist

\subsection*{Second Derivative Test}
\textbf{Hessian:} $D = f_{xx}f_{yy} - (f_{xy})^2$ at $(a,b)$
\begin{itemize}[leftmargin=*,nosep]
\item $D > 0$ and $f_{xx} > 0$: Relative minimum
\item $D > 0$ and $f_{xx} < 0$: Relative maximum
\item $D < 0$: Saddle point
\item $D = 0$: Inconclusive
\end{itemize}

\subsection*{Lagrange Multipliers}
To optimize $f(x,y,z)$ subject to $g(x,y,z) = k$: \\
Solve $\nabla f = \lambda \nabla g$ and $g(x,y,z) = k$

\textbf{Multiple Constraints:} $g_1 = k_1, g_2 = k_2$: \\
$\nabla f = \lambda_1 \nabla g_1 + \lambda_2 \nabla g_2$

\section*{Double Integrals}

\subsection*{Rectangular Regions}
$\iint_R f(x,y) \dd{A} = \int_c^d \int_a^b f(x,y) \dd{x}\dd{y}$

If $f(x,y) = g(x)h(y)$: \\
$\iint_R g(x)h(y) \dd{A} = \left(\int_a^b g(x)\dd{x}\right)\left(\int_c^d h(y)\dd{y}\right)$

\subsection*{General Regions}
\textbf{Type I:} $D = \{(x,y) | a \le x \le b, g_1(x) \le y \le g_2(x)\}$ \\
$\iint_D f(x,y) \dd{A} = \int_a^b \int_{g_1(x)}^{g_2(x)} f(x,y) \dd{y}\dd{x}$

\textbf{Type II:} $D = \{(x,y) | c \le y \le d, h_1(y) \le x \le h_2(y)\}$ \\
$\iint_D f(x,y) \dd{A} = \int_c^d \int_{h_1(y)}^{h_2(y)} f(x,y) \dd{x}\dd{y}$

\subsection*{Polar Coordinates}
$x = r\cos\theta, \quad y = r\sin\theta, \quad \dd{A} = r\dd{r}\dd{\theta}$ \\
$\iint_D f(x,y) \dd{A} = \int_\alpha^\beta \int_{h_1(\theta)}^{h_2(\theta)} f(r\cos\theta,r\sin\theta) r\dd{r}\dd{\theta}$

\textbf{Area:} $A = \iint_D \dd{A}$

\section*{Triple Integrals}

\subsection*{Rectangular Box}
$\iiint_B f(x,y,z) \dd{V} = \int_r^s \int_c^d \int_a^b f(x,y,z) \dd{x}\dd{y}\dd{z}$

\subsection*{General Regions}
\textbf{Type I:} $(x,y) \in D$, $u_1(x,y) \le z \le u_2(x,y)$ \\
$\iiint_E f \dd{V} = \iint_D \left[\int_{u_1}^{u_2} f(x,y,z) \dd{z}\right] \dd{A}$

\textbf{Type II:} $(y,z) \in D$, $u_1(y,z) \le x \le u_2(y,z)$

\textbf{Type III:} $(x,z) \in D$, $u_1(x,z) \le y \le u_2(x,z)$

\subsection*{Cylindrical Coordinates}
$x = r\cos\theta, \quad y = r\sin\theta, \quad z = z$ \\
$\dd{V} = r\dd{z}\dd{r}\dd{\theta}$ \\
$\iiint_E f \dd{V} = \int_\alpha^\beta \int_{h_1}^{h_2} \int_{u_1}^{u_2} f(r\cos\theta,r\sin\theta,z) r\dd{z}\dd{r}\dd{\theta}$

\subsection*{Spherical Coordinates}
$x = \rho\sin\varphi\cos\theta, \quad y = \rho\sin\varphi\sin\theta, \quad z = \rho\cos\varphi$ \\
$x^2 + y^2 + z^2 = \rho^2$ \\
$\dd{V} = \rho^2\sin\varphi\dd{\rho}\dd{\theta}\dd{\varphi}$ \\
$0 \le \theta \le 2\pi, \quad 0 \le \varphi \le \pi$

\subsection*{Change of Variables (Jacobian)}
\textbf{2D:} $\dd{A} = \left|\pdv{(x,y)}{(u,v)}\right| \dd{u}\dd{v}$ where \\
$\pdv{(x,y)}{(u,v)} = \begin{vmatrix} \pdv{x}{u} & \pdv{x}{v} \\ \pdv{y}{u} & \pdv{y}{v} \end{vmatrix}$

\textbf{3D:} $\dd{V} = \left|\pdv{(x,y,z)}{(u,v,w)}\right| \dd{u}\dd{v}\dd{w}$ where \\
$\pdv{(x,y,z)}{(u,v,w)} = \begin{vmatrix} \pdv{x}{u} & \pdv{x}{v} & \pdv{x}{w} \\ \pdv{y}{u} & \pdv{y}{v} & \pdv{y}{w} \\ \pdv{z}{u} & \pdv{z}{v} & \pdv{z}{w} \end{vmatrix}$

\subsection*{Surface Area}
For $z = f(x,y)$ over region $D$: \\
$S = \iint_D \sqrt{1 + \left(\pdv{f}{x}\right)^2 + \left(\pdv{f}{y}\right)^2} \dd{A}$

For parametric surface $\vec{r}(u,v)$ over $D$: \\
$S = \iint_D \|\vec{r}_u \times \vec{r}_v\| \dd{A}$

\section*{Vector Fields}

$\vec{F}(x,y,z) = P\hat{i} + Q\hat{j} + R\hat{k}$ \\
$\vec{F}(x,y) = P\hat{i} + Q\hat{j}$

\textbf{Gradient Field:} $\vec{F} = \nabla f$ \\
\textbf{Conservative:} $\vec{F} = \nabla f$ for some potential $f$

\section*{Line Integrals}

\subsection*{With Respect to Arc Length}
$\int_C f(x,y,z) \dd{s} = \int_a^b f(\vec{r}(t)) \|\vec{r}'(t)\| \dd{t}$

\textbf{Direction independent}

\subsection*{With Respect to Coordinates}
$\int_C f(x,y) \dd{x} = \int_a^b f(x(t),y(t)) x'(t) \dd{t}$ \\
$\int_C f(x,y) \dd{y} = \int_a^b f(x(t),y(t)) y'(t) \dd{t}$

\textbf{Combined:} $\int_C P\dd{x} + Q\dd{y} + R\dd{z}$

\textbf{Direction dependent:} $\int_{-C} f \dd{x} = -\int_C f \dd{x}$

\subsection*{Of Vector Fields}
$\int_C \vec{F} \cdot \dd{\vec{r}} = \int_a^b \vec{F}(\vec{r}(t)) \cdot \vec{r}'(t) \dd{t}$ \\
$= \int_C P\dd{x} + Q\dd{y} + R\dd{z}$ \\
$= \int_C \vec{F} \cdot \vec{T} \dd{s}$

\textbf{Work Interpretation:} Work done by force field $\vec{F}$ along $C$

\subsection*{Conservative Vector Fields}
\textbf{Test ($\R^2$):} $\vec{F} = P\hat{i} + Q\hat{j}$ is conservative if \\
$\pdv{P}{y} = \pdv{Q}{x}$ (on simply connected region)

\textbf{Test ($\R^3$):} $\vec{F}$ is conservative if $\text{curl }\vec{F} = \vec{0}$

\textbf{Finding Potential:} Solve $\pdv{f}{x} = P, \pdv{f}{y} = Q, \pdv{f}{z} = R$

\subsection*{Fundamental Theorem for Line Integrals}
If $\vec{F} = \nabla f$: $\int_C \nabla f \cdot \dd{\vec{r}} = f(\vec{r}(b)) - f(\vec{r}(a))$

Path independent: $\int_C \vec{F} \cdot \dd{\vec{r}} = 0$ for closed $C$

\subsection*{Green's Theorem}
$C$ positively oriented, simple closed curve, $D$ region inside: \\
$\oint_C P\dd{x} + Q\dd{y} = \iint_D \left(\pdv{Q}{x} - \pdv{P}{y}\right) \dd{A}$

\textbf{Vector Form:} $\oint_C \vec{F} \cdot \dd{\vec{r}} = \iint_D (\text{curl }\vec{F}) \cdot \hat{k} \dd{A}$

\textbf{Flux Form:} $\oint_C \vec{F} \cdot \vec{n} \dd{s} = \iint_D (\text{div }\vec{F}) \dd{A}$

\textbf{Area:} $A = \oint_C x\dd{y} = -\oint_C y\dd{x} = \frac{1}{2}\oint_C x\dd{y} - y\dd{x}$

\section*{Curl \& Divergence}

\subsection*{Curl}
$\text{curl }\vec{F} = \nabla \times \vec{F} = \begin{vmatrix} \hat{i} & \hat{j} & \hat{k} \\ \pdv{}{x} & \pdv{}{y} & \pdv{}{z} \\ P & Q & R \end{vmatrix}$ \\
$= \left(\pdv{R}{y} - \pdv{Q}{z}\right)\hat{i} + \left(\pdv{P}{z} - \pdv{R}{x}\right)\hat{j} + \left(\pdv{Q}{x} - \pdv{P}{y}\right)\hat{k}$

\textbf{Properties:} \\
$\text{curl}(\nabla f) = \vec{0}$ \\
If $\text{curl }\vec{F} = \vec{0}$ on $\R^3$, then $\vec{F}$ is conservative

\subsection*{Divergence}
$\text{div }\vec{F} = \nabla \cdot \vec{F} = \pdv{P}{x} + \pdv{Q}{y} + \pdv{R}{z}$

\textbf{Properties:} \\
$\text{div}(\text{curl }\vec{F}) = 0$ \\
$\nabla^2 f = \text{div}(\nabla f) = \pdv[2]{f}{x} + \pdv[2]{f}{y} + \pdv[2]{f}{z}$ (Laplacian)

\section*{Surface Integrals}

\subsection*{Parametric Surfaces}
$\vec{r}(u,v) = x(u,v)\hat{i} + y(u,v)\hat{j} + z(u,v)\hat{k}$

\textbf{Tangent Vectors:} \\
$\vec{r}_u = \pdv{x}{u}\hat{i} + \pdv{y}{u}\hat{j} + \pdv{z}{u}\hat{k}$ \\
$\vec{r}_v = \pdv{x}{v}\hat{i} + \pdv{y}{v}\hat{j} + \pdv{z}{v}\hat{k}$

\textbf{Normal Vector:} $\vec{n} = \vec{r}_u \times \vec{r}_v$

\subsection*{Surface Integral (Scalar)}
For $z = g(x,y)$ over $D$: \\
$\iint_S f(x,y,z) \dd{S} = \iint_D f(x,y,g(x,y)) \sqrt{1 + g_x^2 + g_y^2} \dd{A}$

For parametric $\vec{r}(u,v)$ over $D$: \\
$\iint_S f(x,y,z) \dd{S} = \iint_D f(\vec{r}(u,v)) \|\vec{r}_u \times \vec{r}_v\| \dd{A}$

\subsection*{Surface Integral (Vector Field / Flux)}
$\iint_S \vec{F} \cdot \dd{\vec{S}} = \iint_S \vec{F} \cdot \vec{n} \dd{S}$

For $z = g(x,y)$ (upward orientation): \\
$\iint_S \vec{F} \cdot \dd{\vec{S}} = \iint_D (-Pg_x - Qg_y + R) \dd{A}$

For parametric $\vec{r}(u,v)$: \\
$\iint_S \vec{F} \cdot \dd{\vec{S}} = \iint_D \vec{F}(\vec{r}(u,v)) \cdot (\vec{r}_u \times \vec{r}_v) \dd{A}$

\subsection*{Stokes' Theorem}
$S$ oriented surface, $C$ boundary with positive orientation: \\
$\int_C \vec{F} \cdot \dd{\vec{r}} = \iint_S \text{curl }\vec{F} \cdot \dd{\vec{S}}$

\subsection*{Divergence Theorem}
$E$ solid region, $S$ boundary with outward orientation: \\
$\iint_S \vec{F} \cdot \dd{\vec{S}} = \iiint_E \text{div }\vec{F} \dd{V}$

\section*{Common Parametrizations}

\textbf{Sphere ($x^2+y^2+z^2=a^2$):} \\
$\vec{r}(\theta,\varphi) = a\sin\varphi\cos\theta\hat{i} + a\sin\varphi\sin\theta\hat{j} + a\cos\varphi\hat{k}$ \\
$0 \le \theta \le 2\pi, \quad 0 \le \varphi \le \pi$

\textbf{Cylinder ($x^2+y^2=a^2$):} \\
$\vec{r}(z,\theta) = a\cos\theta\hat{i} + a\sin\theta\hat{j} + z\hat{k}$ \\
$0 \le \theta \le 2\pi$

\textbf{Cone ($z = \sqrt{x^2+y^2}$):} \\
$\vec{r}(r,\theta) = r\cos\theta\hat{i} + r\sin\theta\hat{j} + r\hat{k}$ \\
$0 \le \theta \le 2\pi$
